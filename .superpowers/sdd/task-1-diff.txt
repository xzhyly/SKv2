191:   const coordsVal = document.getElementById('coordsVal');
192:   const consoleBox = document.getElementById('consoleBox');
193:   let activeStream = null;
194:   let hands = null;
195:   let cameraLoop = null;
196:   let lastFrameTime = performance.now();
197:   let fpsHistory = [];
198:   const offscreenCanvas = document.createElement('canvas');
199:   offscreenCanvas.width = 320;
200:   offscreenCanvas.height = 240;
201:   const offscreenCtx = offscreenCanvas.getContext('2d', { willReadFrequently: true });
202:   function log(message, type = 'info') {
203:     const time = new Date().toTimeString().split(' ')[0];
204:     const prefix = type === 'warn' ? '⚠️ ' : type === 'error' ? '❌ ' : 'ℹ️ ';
205:     const logLine = `[${time}] ${prefix}${message}`;
206:     consoleBox.textContent += `\n${logLine}`;
207:     consoleBox.scrollTop = consoleBox.scrollHeight;
208:     console.log(logLine);
209:   }
210:   async function initMediaPipe() {
211:     if (!window.Hands) {
212:       log('MediaPipe Hands is not loaded! Check js/lib/mediapipe/hands.js path.', 'error');
213:       return false;
214:     }
215:     log('Initializing MediaPipe Hands...', 'info');
216:     hands = new window.Hands({
217:       locateFile: (file) => `js/lib/mediapipe/${file}`
218:     });
219:     hands.setOptions({
220:       maxNumHands: 2,
221:       modelComplexity: 1,
222:       minDetectionConfidence: 0.65,
223:       minTrackingConfidence: 0.65
224:     });
225:     hands.onResults(onResults);
226:     log('MediaPipe Hands initialized.', 'info');
227:     return true;
228:   }
229:   function calculateFPS() {
230:     const now = performance.now();
231:     const duration = now - lastFrameTime;
232:     lastFrameTime = now;
233:     if (duration > 0 && duration <= 500) {
234:       const currentFps = 1000 / duration;
235:       fpsHistory.push(currentFps);
236:       if (fpsHistory.length > 10) fpsHistory.shift();
237:       const avgFps = fpsHistory.reduce((a, b) => a + b, 0) / fpsHistory.length;
238:       fpsVal.textContent = avgFps.toFixed(1);
239:     }
240:   }
241:   function onResults(results) {
242:     if (!activeStream) return;
243:     calculateFPS();
244:     const bounds = canvas.getBoundingClientRect();
245:     if (canvas.width !== bounds.width || canvas.height !== bounds.height) {
246:       canvas.width = bounds.width;
247:       canvas.height = bounds.height;
248:     }
249:     ctx.clearRect(0, 0, canvas.width, canvas.height);
250:     const detectedHands = results.multiHandLandmarks || [];
251:     if (detectedHands.length === 0) {
252:       stateVal.textContent = 'No hand detected';
253:       coordsVal.textContent = 'x: 0, y: 0, z: 0';
254:       return;
255:     }
256:     stateVal.textContent = `${detectedHands.length} hand(s) detected`;
257:     detectedHands.forEach((landmarks) => {
258:       const indexTip = landmarks[8];
259:       coordsVal.textContent = `x: ${indexTip.x.toFixed(3)}, y: ${indexTip.y.toFixed(3)}, z: ${indexTip.z.toFixed(3)}`;
260:       const connections = [
261:         [0, 1], [1, 2], [2, 3], [3, 4],
262:         [0, 5], [5, 6], [6, 7], [7, 8],
263:         [0, 9], [9, 10], [10, 11], [11, 12],
264:         [0, 13], [13, 14], [14, 15], [15, 16],
265:         [0, 17], [17, 18], [18, 19], [19, 20],
266:         [5, 9], [9, 13], [13, 17]
267:       ];
268:       ctx.lineWidth = 4;
269:       ctx.strokeStyle = '#10b981';
270:       connections.forEach(([start, end]) => {
271:         const pt1 = landmarks[start];
272:         const pt2 = landmarks[end];
273:         ctx.beginPath();
274:         ctx.moveTo(pt1.x * canvas.width, pt1.y * canvas.height);
275:         ctx.lineTo(pt2.x * canvas.width, pt2.y * canvas.height);
276:         ctx.stroke();
277:       });
278:       ctx.fillStyle = '#ef4444';
279:       landmarks.forEach((point) => {
280:         ctx.beginPath();
281:         ctx.arc(point.x * canvas.width, point.y * canvas.height, 6, 0, 2 * Math.PI);
282:         ctx.fill();
283:       });
284:     });
285:   }
286:   async function runCameraLoop() {
287:     if (!hands || !activeStream) return;
288:     if (video.readyState >= 2) {
289:       try {
290:         const sw = video.videoWidth;
291:         const sh = video.videoHeight;
292:         if (sw && sh) {
293:           const targetAspect = 4 / 3;
294:           const currentAspect = sw / sh;
295:           let sx = 0, sy = 0, sWidth = sw, sHeight = sh;
296:           if (currentAspect > targetAspect) {
297:             sWidth = sh * targetAspect;
298:             sx = (sw - sWidth) / 2;
299:           } else if (currentAspect < targetAspect) {
300:             sHeight = sw / targetAspect;
301:             sy = (sh - sHeight) / 2;
302:           }
303:           offscreenCtx.drawImage(video, sx, sy, sWidth, sHeight, 0, 0, 320, 240);
304:         } else {
305:           offscreenCtx.drawImage(video, 0, 0, 320, 240);
306:         }
307:         await hands.send({ image: offscreenCanvas });
308:       } catch (err) {
309:         log('MediaPipe send frame error: ' + err.message, 'error');
310:       }
311:     }
312:     cameraLoop = requestAnimationFrame(runCameraLoop);
313:   }
314:   async function startCapture() {
315:     log('Starting camera capture...', 'info');
316:     startBtn.disabled = true;
317:     if (!hands) {
318:       const ok = await initMediaPipe();
319:       if (!ok) {
320:         startBtn.disabled = false;
321:         return;
322:       }
323:     }
324:     try {
325:       activeStream = await navigator.mediaDevices.getUserMedia({
326:         video: { width: { ideal: 640 }, height: { ideal: 480 } }
327:       });
328:       video.srcObject = activeStream;
329:       video.play().catch(e => log('Video play failed: ' + e.message, 'warn'));
330:       log('Webcam connected successfully.', 'info');
331:       stopBtn.disabled = false;
332:       lastFrameTime = performance.now();
333:       runCameraLoop();
334:     } catch (err) {
335:       log('Camera access failed: ' + err.message, 'error');
336:       startBtn.disabled = false;
337:     }
338:   }
339:   function stopCapture() {
340:     log('Stopping camera capture...', 'info');
341:     stopBtn.disabled = true;
342:     if (cameraLoop) {
343:       cancelAnimationFrame(cameraLoop);
344:       cameraLoop = null;
345:     }
346:     if (activeStream) {
347:       activeStream.getTracks().forEach(track => track.stop());
348:       activeStream = null;
349:     }
350:     video.srcObject = null;
351:     ctx.clearRect(0, 0, canvas.width, canvas.height);
352:     fpsVal.textContent = '0.0';
353:     stateVal.textContent = 'No hand detected';
354:     coordsVal.textContent = 'x: 0, y: 0, z: 0';
355:     startBtn.disabled = false;
356:     log('Capture stopped.', 'info');
357:   }
358:   startBtn.addEventListener('click', startCapture);
359:   stopBtn.addEventListener('click', stopCapture);
